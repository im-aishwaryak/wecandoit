/**
 * @file report_item.js
 * @description Handles submission of new lost/found item reports.
 *
 * This module allows authenticated users to submit item reports
 * to Firestore, including item details, images, and verification
 * questions. Images are uploaded to Cloudinary before being stored
 * in the database.
 *
 * Features:
 * - Form validation via Firebase auth state
 * - Image upload to Cloudinary
 * - Dynamic verification question storage
 * - Firestore document creation
 * - Submission status tracking
 *
 * @module reportItem
 * @author Aadhya Goyal
 * @version 1.0
 */

import {
    db, auth, provider, signInWithPopup, signOut,
    doc, getDoc, setDoc, collection, addDoc, updateDoc, deleteDoc,
    deleteField, storage, ref, uploadBytes, getDownloadURL
} from './firebaseModule.js';



/**
 * Submit button element for reporting an item.
 *
 * Triggers the item upload process when clicked.
 *
 * @type {HTMLElement}
 */
const report_button = document.getElementById("submit-button");

/**
 * Uploads a new item report to Firestore.
 *
 * Collects form data from the report item page,
 * uploads the image to Cloudinary, and stores
 * the full item record in the "Item_Data" collection.
 *
 * Also attaches:
 * - Verification questions and answers
 * - Submitter authentication data
 * - Timestamp metadata
 * - Initial item status ("pending submission")
 *
 * @async
 * @function uploadData
 * @returns {Promise<void>}
 */
const uploadData = async() => {
    console.log("in function")
   
    // Get the current user
    const currentUser = auth.currentUser;
   
    if (!currentUser) {
        console.error("No user is logged in!");
        alert("You must be logged in to submit an item.");
        return;
    }
   
    var category = document.getElementById("category").value;
    var item_name = document.getElementById("item").value;
    var public_desc = document.getElementById("public_desc").value;
    var private_desc = document.getElementById("private_desc").value;
    var location = document.getElementById("location").value;
    var date = document.getElementById("date").value;
    var time = document.getElementById("time").value;


    var image_url = await uploadImage()


    var q1 = document.getElementById("question1").value
    var a1 = document.getElementById("answer1").value
   
    var q2 = document.getElementById("question2").value
    var a2 = document.getElementById("answer2").value


    var q3 = document.getElementById("question3").value
    var a3 = document.getElementById("answer3").value


    var verifications = {[q1]:a1, [q2]:a2, [q3]:a3}
    console.log(verifications)


    var ref = collection(db, "Item_Data")
    console.log(ref)
    console.log(localStorage.getItem("user_logged_in"))
   
    await addDoc(
        ref, {
            category: category,
            item_name: item_name,
            image_url: image_url,
            public_description: public_desc,
            private_description: private_desc,
            location_found: location,
            date_found: date,
            time_found: time,
            verification_qs: verifications,
            status: "pending submission",
            retrieval_date: "",
            // Add user tracking fields
            submitter_id: currentUser.uid,
            submitter_email: currentUser.email,
            submitted_at: new Date().toISOString()
        }
    );
}




/**
 * Handles item report submission.
 *
 * Prevents page reload, uploads form data,
 * stores item in Firestore, and refreshes
 * the page after successful submission.
 *
 * @listens click
 */
report_button.addEventListener("click", async (e) => {
    e.preventDefault();   // 🔥 stops page reload
    await uploadData();
    alert("Item Report Submitted!")
    window.location.reload();
});




/**
 * Uploads an image file to Cloudinary.
 *
 * Sends the selected image file to the Cloudinary
 * API and retrieves a secure URL for storage
 * in Firestore.
 *
 * @async
 * @function uploadImage
 * @returns {Promise<string>}
 * Secure URL of the uploaded image.
 */
const uploadImage = async() => {
    const data = new FormData();
    const file = document.getElementById("image_file").files[0];
    data.append("file", file);
    data.append("upload_preset", "boomerang_uploads");


    const res = await fetch(
        "https://api.cloudinary.com/v1_1/dpj2xj3ry/image/upload",
        {
            method: "POST",
            body: data
        }
    );


    const json = await res.json();
    return json.secure_url; // THIS is the image URL we store in Firestore
}
