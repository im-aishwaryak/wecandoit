/**
 * @file auth.js
 * @brief Handles user authentication and account creation.
 *
 * This script manages:
 * - Google sign-in authentication
 * - Email/password account creation
 * - User role assignment (Student or Admin)
 * - Firestore user document creation
 * - Redirecting users to appropriate dashboards
 *
 * User roles are determined based on email domain:
 * - "@apps.nsd.org" → Student
 * - "@nsd.org" → Admin
 *
 * After authentication, a Firestore document is created
 * in the "User_Data" collection to store user metadata.
 *
 * @author Aadhya Goyal, Aanya Rawal, Aishwarya Kumaran
 * @version 1.0
 */





import {
    db, auth, provider, signInWithPopup, createUserWithEmailAndPassword, signOut,
    doc, getDoc, setDoc, collection, addDoc, updateDoc, deleteDoc,
    deleteField, storage, ref, uploadBytes, getDownloadURL
} from './firebaseModule.js';






/**

let user;


/**
 * Stores the role of the authenticated user.
 * Possible values:
 * - "Student"
 * - "Admin"
 * @type {string}

let user_status;

/**

    signInWithPopup(auth, provider)

/**
 * Handles email/password user registration.
 *
 * This function:
 * 1. Reads email and password from input fields
 * 2. Creates a Firebase authentication account
 * 3. Adds a Firestore user document
 * 4. Stores user identity in localStorage
 * 5. Redirects the user to the correct dashboard
 *
 * @async
 * @function userEmailSignIn
 */
const userEmailSignIn = async () => {
    var email = document.getElementById("email").value;
    var password = document.getElementById("password").value;


    createUserWithEmailAndPassword(auth, email, password)
  .then(async (userCredential) => {
    // Signed up
    user = userCredential.user;
    await AddUser()
    if(user_status == "Student"){
      localStorage.setItem("user_identity", "student")
      window.location.href = "dashboard.html";
  }
    else{
      localStorage.setItem("user_identity", "admin")
      window.location.href = "admin-dashboard.html"
    }
  })
  .catch((error) => {
    const errorCode = error.code;
    const errorMessage = error.message;
    alert(errorMessage)
  });
}

/**
 * Creates or updates a Firestore user document.
 *
 * This function:
 * 1. Validates the authenticated user
 * 2. Determines user role from email domain
 * 3. Creates a Firestore document in "User_Data"
 * 4. Stores default user metadata
 *
 * Role rules:
 * - "@apps.nsd.org" → Student
 * - "@nsd.org" → Admin
 *
 * @async
 * @function AddUser
 */
async function AddUser() {
    if (!user) {
        console.log("no user ")
        return;
    }
    if (auth.currentUser.email.includes("@apps.nsd.org")){
      user_status = "Student"
    }
    else if (auth.currentUser.email.includes("@nsd.org")){
      user_status = "Admin"
    }
    else{
      alert("email must be tied to North Creek High School")
      return;
    }
    var ref = doc(db, "User_Data", user.email);
    await setDoc(
      ref, {
      items_posted: 0,
      status: user_status
    })
}
