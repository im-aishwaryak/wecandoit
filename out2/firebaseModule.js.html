/**
 * Retrieves all items from the database.
 *
 * Fetches all documents from the
 * "Item_Data" collection and converts
 * them into plain JavaScript objects.
 *
 * @async
 * @function getAllItems
 * @returns {Promise<Array<Object>>}
 * Array of item objects containing
 * document IDs and data fields.
 */
export async function getAllItems() {
  // NOTE: use the exact collection name from your console
  const ref = collection(db, "Item_Data");
  const snap = await getDocs(ref);

  // Map docs to plain objects
  return snap.docs.map(d => ({
    id: d.id,
    ...d.data()
  }));
}

/**
 * Creates a real-time listener
 * for item updates.
 *
 * Listens for changes in the
 * "Item_Data" collection and
 * sends updated item data to
 * the provided callback function.
 *
 * @function listenToItems
 * @param {Function} callback
 * Function executed when data changes.
 *
 * @returns {Function}
 * Unsubscribe function used
 * to stop listening for updates.
 */
export function listenToItems(callback) {
  return onSnapshot(collection(db, "Item_Data"), snap => {
    const items = snap.docs.map(d => ({ id: d.id, ...d.data() }));
    callback(items);
  });
}

/**
 * Retrieves a single item by ID.
 *
 * Fetches the specified document
 * from the "Item_Data" collection.
 *
 * Returns null if the item does not exist.
 *
 * @async
 * @function getItemById
 * @param {string} id - Unique item ID.
 *
 * @returns {Promise<Object|null>}
 * Item object if found,
 * otherwise null.
 */
export async function getItemById(id) {
  const docRef = doc(db, "Item_Data", id);
  const docSnap = await getDoc(docRef);
  if (!docSnap.exists()) return null;
  return { id: docSnap.id, ...docSnap.data() };
}
